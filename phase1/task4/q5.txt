Since "Add Task" lives inside the same task1/backend app as every other endpoint, there's no separate deployment unit for it — the Dockerfile and CI config package the whole FastAPI service, and the POST /tasks route just comes along for free.

Dockerfile:


FROM python:3.12-slim
WORKDIR /app
COPY requirements.txt .
RUN pip install --no-cache-dir -r requirements.txt
COPY . .
EXPOSE 8000
CMD ["uvicorn", "main:app", "--host", "0.0.0.0", "--port", "8000"]
CI/CD config (.github/workflows/ci.yml):


name: Backend CI

on:
  push:
    paths:
      - "phase1/task1/backend/**"

jobs:
  test:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-python@v5
        with:
          python-version: "3.12"
      - name: Install dependencies
        working-directory: phase1/task1/backend
        run: |
          pip install -r requirements.txt
          pip install pytest httpx
      - name: Run tests
        working-directory: phase1/task1/backend
        run: pytest

  build:
    needs: test
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - name: Build Docker image
        run: |
          docker build -f phase1/task4/artifacts/Dockerfile \
            -t task-manager-api:${{ github.sha }} \
            phase1/task1/backend
